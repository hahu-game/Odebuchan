# 3.2 UIController.cs (行動選択ボタンの実装)

## ファイルパス
`Assets/Scripts/UI/UIController.cs`

## 目的
プレイヤーが行動を選択できるようにする

## コード内容（既存のUIController.csに追加）

```csharp
// 既存のUIController.csに以下を追加

using UnityEngine.UI;

public class UIController : MonoBehaviour
{
    // === 既存の変数 ===
    // (既存のコードはそのまま)

    // === 新規追加UI参照 ===
    [Header("行動選択ボタン")]
    public Button eatButton;
    public Button sleepButton;
    public Button playButton;
    public Button clinicButton;

    [Header("操作ボタン")]
    public Button fixButton;
    public Button clearButton;

    // === ローカル変数（選択状態） ===
    private ActionType? _selectedMorningAction = null;
    private ActionType? _selectedAfternoonAction = null;
    private bool _isSelectingMorning = true; // true = 午前選択中、false = 午後選択中

    // 自分のPlayerActionDataへの参照
    private PlayerActionData _myPlayerActionData;

    private void Start()
    {
        // ボタンのクリックイベントを設定
        eatButton.onClick.AddListener(() => OnActionButtonClicked(ActionType.Eat));
        sleepButton.onClick.AddListener(() => OnActionButtonClicked(ActionType.Sleep));
        playButton.onClick.AddListener(() => OnActionButtonClicked(ActionType.Play));
        clinicButton.onClick.AddListener(() => OnActionButtonClicked(ActionType.Clinic));

        fixButton.onClick.AddListener(OnFixButtonClicked);
        clearButton.onClick.AddListener(OnClearButtonClicked);

        // 初期状態では行動選択ボタンを無効化
        SetActionButtonsInteractable(false);
    }

    /// <summary>
    /// 自分のPlayerActionDataを設定（GameManagerから呼び出す）
    /// </summary>
    public void SetMyPlayerActionData(PlayerActionData actionData)
    {
        _myPlayerActionData = actionData;
    }

    /// <summary>
    /// 行動ボタンがクリックされた時の処理
    /// </summary>
    public void OnActionButtonClicked(ActionType action)
    {
        if (_isSelectingMorning)
        {
            // 午前の行動を設定
            _selectedMorningAction = action;
            Debug.Log($"午前の行動を選択: {action}");

            // 午後の選択に移行
            _isSelectingMorning = false;

            // 視覚的なフィードバック（後で実装）
            // UpdateActionPanelHighlight();
        }
        else
        {
            // 午後の行動を設定
            _selectedAfternoonAction = action;
            Debug.Log($"午後の行動を選択: {action}");

            // 視覚的なフィードバック（後で実装）
            // UpdateActionPanelHighlight();
        }

        // 選択状態をログに表示
        Debug.Log($"現在の選択: 午前={_selectedMorningAction}, 午後={_selectedAfternoonAction}");
    }

    /// <summary>
    /// 確定ボタンがクリックされた時の処理
    /// </summary>
    public void OnFixButtonClicked()
    {
        // 午前・午後が両方選択されているか確認
        if (!_selectedMorningAction.HasValue || !_selectedAfternoonAction.HasValue)
        {
            Debug.LogWarning("午前と午後の両方を選択してください");
            AddLog("午前と午後の両方を選択してください");
            return;
        }

        // 選択した行動をPlayerActionDataに送信
        if (_myPlayerActionData != null)
        {
            ActionData morningAction = CreateActionData(_selectedMorningAction.Value);
            ActionData afternoonAction = CreateActionData(_selectedAfternoonAction.Value);

            _myPlayerActionData.RPC_SetMorningAction(morningAction);
            _myPlayerActionData.RPC_SetAfternoonAction(afternoonAction);
            _myPlayerActionData.RPC_FixActions();

            Debug.Log("行動を確定しました");
            AddLog("行動を確定しました");

            // 確定後、ボタンを無効化
            SetActionButtonsInteractable(false);
            fixButton.interactable = false;
            clearButton.interactable = false;
        }
        else
        {
            Debug.LogError("PlayerActionData が設定されていません");
        }
    }

    /// <summary>
    /// クリアボタンがクリックされた時の処理
    /// </summary>
    public void OnClearButtonClicked()
    {
        // 選択をリセット
        _selectedMorningAction = null;
        _selectedAfternoonAction = null;
        _isSelectingMorning = true;

        Debug.Log("選択をクリアしました");

        // RPCでサーバー側もクリア
        if (_myPlayerActionData != null)
        {
            _myPlayerActionData.RPC_ClearActions();
        }

        // 視覚的なフィードバック（後で実装）
        // UpdateActionPanelHighlight();
    }

    /// <summary>
    /// ActionTypeからActionDataを生成するヘルパーメソッド
    /// </summary>
    private ActionData CreateActionData(ActionType type)
    {
        return type switch
        {
            ActionType.Eat => ActionData.CreateEat(),
            ActionType.Sleep => ActionData.CreateSleep(),
            ActionType.Play => ActionData.CreatePlay(),
            ActionType.Clinic => ActionData.CreateClinic(),
            _ => ActionData.Default()
        };
    }

    /// <summary>
    /// 行動選択ボタンの操作可否を設定
    /// </summary>
    public void SetActionButtonsInteractable(bool interactable)
    {
        eatButton.interactable = interactable;
        sleepButton.interactable = interactable;
        playButton.interactable = interactable;
        clinicButton.interactable = interactable;
        fixButton.interactable = interactable;
        clearButton.interactable = interactable;
    }

    /// <summary>
    /// 選択フェーズ開始時に呼び出す（GameFlowManagerから）
    /// </summary>
    public void OnSelectionPhaseStart()
    {
        // 選択をリセット
        _selectedMorningAction = null;
        _selectedAfternoonAction = null;
        _isSelectingMorning = true;

        // ボタンを有効化
        SetActionButtonsInteractable(true);

        Debug.Log("選択フェーズが開始されました。行動を選択してください。");
        AddLog("行動を選択してください");
    }

    // === 既存のメソッドはそのまま ===
    // UpdateWeightDisplay(), UpdateEnergyDisplay(), etc.

    // === 新規追加メソッド（スタブ） ===

    /// <summary>
    /// ブラックアウトパネルを表示（後で実装）
    /// </summary>
    public void ShowBlackout(string text, float duration)
    {
        Debug.Log($"[ブラックアウト] {text}");
        // 実装は後で
    }

    /// <summary>
    /// ログエリアにメッセージを追加（後で実装）
    /// </summary>
    public void AddLog(string message)
    {
        Debug.Log($"[ログ] {message}");
        // 実装は後で
    }

    // その他、後で実装するメソッドのスタブ
    public void UpdatePlayBuffDisplay(PlayerRef player, int weightBuff, int energyBuff) { }
    public void UpdateEvolutionDisplay(PlayerRef player, bool hasEvolved, string abilityName) { }
    public void UpdateUyopyonVisual(PlayerRef player, string visualType) { }
    public void UpdateStatusAilmentDisplay(PlayerRef player, byte[] ailments) { }
}
```

```csharp
// SpawnPlayerActionData()の中または後で
if (player == Runner.LocalPlayer)

    // 自分のPlayerActionDataをUIControllerに設定

}
```

## 確認項目
- [ ] たべるボタンをクリックすると午前の選択として記録されること
- [ ] もう一度別のボタンをクリックすると午後の選択として記録されること
- [ ] クリアボタンで選択がリセットされること
- [ ] 確定ボタンで選択がネットワーク送信されること（Debug.Logで確認）
- [ ] 確定後、ボタンが無効化されること

- 視覚的なフィードバック（パネルの太枠表示など）は次の段階で実装
