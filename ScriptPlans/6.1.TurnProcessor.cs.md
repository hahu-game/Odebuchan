# 6.1 TurnProcessor.cs (状態異常の発症処理 - 病気)

## ファイルパス
`Assets/Scripts/Game/TurnProcessor.cs`

## 目的
たべる実行時の病気発症判定を実装する

## 実装内容

### CheckSickness()メソッドの追加

```csharp
/// <summary>
/// 病気の発症判定（たべる実行時）
/// </summary>
/// <param name="player">プレイヤー</param>
/// <param name="weight">現在の重さ</param>
private void CheckSickness(PlayerRef player, int weight)
{
    var uyopyon = GameManager.Instance.GetUyopyonState(player);
    if (uyopyon == null) return;

    // 発症確率 = 現在の重さ ÷ 10 (%)
    float sicknessChance = weight / 10f;
    float randomValue = Random.Range(0f, 100f);

    if (randomValue < sicknessChance)
    {
        // 病気発症
        bool hasSleepApnea = HasStatusAilment(uyopyon, StatusAilment.SleepApnea);
        bool hasDiabetes = HasStatusAilment(uyopyon, StatusAilment.Diabetes);

        StatusAilment newAilment;

        if (hasSleepApnea && hasDiabetes)
        {
            // 両方すでに発症している場合は何もしない
            return;
        }
        else if (hasSleepApnea)
        {
            // 睡眠時無呼吸症候群がある場合は糖尿病を発症
            newAilment = StatusAilment.Diabetes;
        }
        else if (hasDiabetes)
        {
            // 糖尿病がある場合は睡眠時無呼吸症候群を発症
            newAilment = StatusAilment.SleepApnea;
        }
        else
        {
            // どちらもない場合は50%ずつで選択
            newAilment = Random.Range(0f, 1f) < 0.5f ? StatusAilment.SleepApnea : StatusAilment.Diabetes;
        }

        // 状態異常を追加
        AddStatusAilment(uyopyon, newAilment);

        // ログに追加
        string ailmentName = GetStatusAilmentName(newAilment);
        GameFlowManager.Instance.RPC_AddLog($"{uyopyon.OwnerPlayer}は{ailmentName}を発症した！");

        Debug.Log($"[TurnProcessor] {player} {ailmentName}発症 (確率: {sicknessChance}%, 判定値: {randomValue})");
    }
}

/// <summary>
/// 状態異常を持っているかチェック
/// </summary>
private bool HasStatusAilment(UyopyonState uyopyon, StatusAilment ailment)
{
    for (int i = 0; i < uyopyon.StatusAilments.Length; i++)
    {
        if (uyopyon.StatusAilments[i] == (byte)ailment)
        {
            return true;
        }
    }
    return false;
}

/// <summary>
/// 状態異常を追加
/// </summary>
private void AddStatusAilment(UyopyonState uyopyon, StatusAilment ailment)
{
    for (int i = 0; i < uyopyon.StatusAilments.Length; i++)
    {
        if (uyopyon.StatusAilments[i] == 0) // 空きスロット
        {
            uyopyon.StatusAilments.Set(i, (byte)ailment);
            return;
        }
    }
}

/// <summary>
/// 状態異常名を取得
/// </summary>
private string GetStatusAilmentName(StatusAilment ailment)
{
    switch (ailment)
    {
        case StatusAilment.SleepApnea: return "睡眠時無呼吸症候群";
        case StatusAilment.Diabetes: return "糖尿病";
        case StatusAilment.BackPain: return "腰痛";
        case StatusAilment.Heatstroke: return "熱中症";
        default: return "";
    }
}
```

### ExecuteAction()でのたべる実行時に呼び出し

```csharp
private void ExecuteAction(PlayerRef player, ActionData action, bool isMorning)
{
    var uyopyon = GameManager.Instance.GetUyopyonState(player);
    if (uyopyon == null) return;

    switch (action.ActionType)
    {
        case ActionType.Eat:
            // たべるの効果
            uyopyon.Energy -= 15;
            uyopyon.Weight += 50 + uyopyon.PlayBuffWeight;

            // 病気発症判定
            CheckSickness(player, uyopyon.Weight);
            break;

        // ... 他の行動 ...
    }
}
```

## 確認項目
- [ ] たべる実行時に病気発症判定が行われること
- [ ] 重さ100の場合、約10%の確率で病気が発症すること
- [ ] 発症した状態異常がログに表示されること
- [ ] すでに睡眠時無呼吸症候群がある場合、2回目の発症で糖尿病になること
- [ ] すでに糖尿病がある場合、2回目の発症で睡眠時無呼吸症候群になること
- [ ] 両方発症済みの場合、それ以上発症しないこと

## 注意点
- UyopyonState.StatusAilmentsはNetworkArray<byte>で最大4つまで
- StatusAilment列挙型をbyteにキャストして格納
- 発症確率は重さに比例（重さ÷10 = 発症確率%）
- ランダム判定はホスト側のみで実行（Networkedプロパティで同期）
- UIへの表示は6.7で実装予定
